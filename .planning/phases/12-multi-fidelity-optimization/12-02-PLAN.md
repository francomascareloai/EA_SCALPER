    Create the sensitivity scoring module with ROBUST math:

    ```python
    from dataclasses import dataclass
    from typing import Optional

    # Epsilon to prevent division by zero
    EPS = 1e-10

    def symmetric_relative_diff(a: float, b: float) -> float:
        """
        Symmetric relative difference: |a - b| / max(|a|, |b|, EPS)

        Properties:
        - Always in [0, 2] range (0 = identical, 2 = opposite signs)
        - Symmetric: srd(a, b) == srd(b, a)
        - Handles zero and negative values correctly
        - No division by zero

        Examples:
        - srd(100, 110) = 10 / 110 = 0.09 (9% difference)
        - srd(100, -100) = 200 / 100 = 2.0 (maximum difference)
        - srd(0, 0) = 0 (identical)
        - srd(0, 10) = 10 / 10 = 1.0 (100% difference)
        """
        diff = abs(a - b)
        max_abs = max(abs(a), abs(b), EPS)
        return diff / max_abs

    @dataclass
    class StrideSensitivityScore:
        """Measures how much a config's metrics change across stride resolutions."""

        pnl_srd: float           # Symmetric relative diff for PnL
        trades_srd: float        # Symmetric relative diff for trades
        max_dd_srd: float        # Symmetric relative diff for max DD
        pnl_sign_flip: bool      # True if PnL changed sign (CRITICAL)

        @property
        def composite_score(self) -> float:
            """
            Weighted composite score.

            Scale: 0.0 = identical, 1.0 = completely different, 2.0 = opposite
            Threshold: > 0.5 (50% relative difference) is suspicious
            """
            base = (
                0.4 * self.pnl_srd +
                0.3 * self.trades_srd +
                0.3 * self.max_dd_srd
            )
            # Sign flip is automatic disqualification
            if self.pnl_sign_flip:
                return 2.0  # Maximum suspicion
            return base

        @property
        def is_suspicious(self) -> bool:
            """Flag configs that change too much between strides."""
            return self.composite_score > 0.5 or self.pnl_sign_flip

        @property
        def disqualification_reason(self) -> Optional[str]:
            """Return reason for disqualification, or None if OK."""
            if self.pnl_sign_flip:
                return "PnL sign flipped between strides (profit ↔ loss)"
            if self.composite_score > 0.5:
                return f"High stride sensitivity: {self.composite_score:.2f} > 0.5"
            return None
    ```

    WHY symmetric_relative_diff:
    - Handles negative PnL correctly (percent change is undefined for negative base)
    - No division by zero (uses epsilon)
    - Bounded output [0, 2] makes thresholds interpretable
    - Symmetric: doesn't matter which stride is "base"

    WHY pnl_sign_flip:
    - If coarse stride shows profit but fine stride shows loss, the config
      is completely unreliable and must be immediately disqualified.
