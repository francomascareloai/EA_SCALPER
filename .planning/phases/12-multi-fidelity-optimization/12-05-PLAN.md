        def promote(self, stage: FidelityStage, results: pd.DataFrame) -> list[StrategyConfig]:
            """Select top configs, excluding high-sensitivity ones."""
            # First: calculate sensitivity for all candidates
            if self.sensitivity_enabled and stage in [FidelityStage.STAGE_1_FILTER, FidelityStage.STAGE_2_REFINE]:
                for config in candidates:
                    score = self.sensitivity_scorer.score_config(config, self.periods, self.catalog_path)

                        results.loc[config.id, "disqualified"] = True

            # Then: rank and promote non-disqualified
            eligible = results[~results.get("disqualified", False)]
            ...

    Add overfitting validation for final stage finalists:

    ```python
    from nautilus_gold_scalper.src.validation.core.config import PBOConfig
    from nautilus_gold_scalper.src.validation.pbo import compute_pbo

    class MultiFidelityOptimizer:
        def __init__(
            self,
            ...,
            pbo_enabled: bool = True,
            pbo_threshold: float = 0.25,  # CLAUDE.md threshold
        ):
            self.pbo_enabled = pbo_enabled
            self.pbo_threshold = pbo_threshold

        def run_tournament(self) -> pd.DataFrame:
            """Run full multi-fidelity tournament with PBO validation."""
            # ... existing stage progression ...

            # After stage 3, compute PBO for finalists
            if self.pbo_enabled:
                finalists_df = self.results[FidelityStage.STAGE_3_VALIDATE]
                for config_id in finalists_df["config_id"]:
                    pbo_score = compute_pbo(
                        config_id=config_id,
                        results=self._get_all_fold_results(config_id),
                    )
                    if pbo_score >= self.pbo_threshold:
                        finalists_df.loc[
                            finalists_df["config_id"] == config_id,
                            "pbo_disqualified"
                        ] = True
                        finalists_df.loc[
                            finalists_df["config_id"] == config_id,
                            "pbo_score"
                        ] = pbo_score
                        self.logger.warning(
                            f"Config {config_id} disqualified: PBO={pbo_score:.1%} >= {self.pbo_threshold:.1%}"
                        )

            return self.results[FidelityStage.STAGE_3_VALIDATE]
    ```

    WHY PBO gate: ARGUS research confirms CSCV+PBO is industry standard for detecting
    overfitting from multiple testing. Without this, large grids will produce
    "winners" that are just statistical noise.

    Threshold: PBO < 25% per CLAUDE.md ml_validation.approval_gate
  </action>
