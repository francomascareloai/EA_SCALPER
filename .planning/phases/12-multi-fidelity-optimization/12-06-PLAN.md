    ```python
    """Validate multi-fidelity optimization configuration before running."""

    def validate_config(config_path: str) -> ValidationResult:
        """Run all pre-flight checks on optimization config."""
        checks = [
            check_parameter_grid_size,      # Warn if >5000 configs
            check_validation_periods,        # Must have >=3 periods, mix of regimes
            check_catalog_exists,            # Verify data catalog accessible
            check_disk_space,                # Need ~1GB per 1000 configs
            check_memory_estimate,           # Warn if expected RAM usage >16GB
            check_stride_correlation_done,   # Warn if 12-01 not completed
            check_sensitivity_thresholds,    # Validate sensitivity config
        ]

        results = [check(config) for check in checks]
        return ValidationResult(
            passed=all(r.passed for r in results),
            warnings=[r.warning for r in results if r.warning],
            errors=[r.error for r in results if r.error],
        )

    if __name__ == "__main__":
        import argparse
        parser = argparse.ArgumentParser()
        parser.add_argument("config", help="Path to optimization config YAML")
        args = parser.parse_args()

        result = validate_config(args.config)
        if result.passed:
            print("✅ All pre-flight checks passed")
        else:
            print("❌ Pre-flight checks failed:")
            for error in result.errors:
                print(f"  ERROR: {error}")
        sys.exit(0 if result.passed else 1)
    ```

    Usage:
    ```bash
    .venv/bin/python -m nautilus_gold_scalper.scripts.optimization.validate_mf_config configs/grids/my_optimization.yaml
    ```
